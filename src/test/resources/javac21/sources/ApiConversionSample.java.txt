import java.util.AbstractMap;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import java.util.TreeSet;
import java.util.function.BinaryOperator;
import java.util.function.Predicate;
import java.util.stream.Collectors;
import java.util.stream.Stream;

/**
 * The Java 9+ APIs Lowbyte can rebuild faithfully.
 *
 * Everything asserted here is contract, not implementation. Set.of and Map.of
 * document their iteration order as unspecified, so the order is never printed;
 * what is printed is what the contract promises, which is rejection of nulls
 * and duplicates, refusal to mutate, and equality with an equivalent collection.
 */
public class ApiConversionSample {

    /**
     * A Collection that ignores toArray's promise to return a fresh array.
     *
     * copyOf accepts any Collection, so it cannot assume that promise is kept.
     * Null-checking an array the source still holds does not stick: the source
     * can drop a null in afterwards. The JDK copies before checking for exactly
     * this reason, calling the race TOCTOU in its own comment.
     */
    private static final class Leaky extends java.util.AbstractList<String> {
        final Object[] shared = { "a", "b" };

        public String get(int index) { return (String) shared[index]; }

        public int size() { return shared.length; }

        @Override public Object[] toArray() { return shared; }
    }

    private static String attempt(Runnable r) {
        try { r.run(); return "ok"; }
        catch (Throwable t) { return t.getClass().getSimpleName(); }
    }

    /**
     * The exception message as well as its type.
     *
     * Objects.checkIndex and friends promise particular wording, and three
     * methods that all throw IndexOutOfBoundsException are only told apart by it.
     */
    private static String detail(Runnable r) {
        try { r.run(); return "ok"; }
        catch (Throwable t) { return t.getClass().getSimpleName() + ": " + t.getMessage(); }
    }

    /** Sorted, so an unspecified iteration order cannot leak into the output. */
    private static String sorted(Iterable<String> values) {
        return new TreeSet<>((java.util.Collection<String>) values).toString();
    }

    public static String runAll() {
        List<String> out = new ArrayList<>();

        List<String> list = List.of("b", "a", "b");
        out.add("list=" + list);
        out.add("list.size=" + list.size());
        out.add("list.equals=" + list.equals(java.util.Arrays.asList("b", "a", "b")));
        out.add("list.add=" + attempt(() -> list.add("c")));
        out.add("list.set=" + attempt(() -> list.set(0, "c")));
        out.add("list.ofNull=" + attempt(() -> List.of((String) null)));
        out.add("list.empty=" + List.of());

        // Every arity has its own overload and so its own descriptor to get
        // right. One wrong descriptor means that arity is silently left alone,
        // which is a NoSuchMethodError on the target and nothing before it.
        out.add("listOf.arities=" + List.of() + List.of("a") + List.of("a", "b")
            + List.of("a", "b", "c") + List.of("a", "b", "c", "d")
            + List.of("a", "b", "c", "d", "e") + List.of("a", "b", "c", "d", "e", "f")
            + List.of("a", "b", "c", "d", "e", "f", "g")
            + List.of("a", "b", "c", "d", "e", "f", "g", "h")
            + List.of("a", "b", "c", "d", "e", "f", "g", "h", "i")
            + List.of("a", "b", "c", "d", "e", "f", "g", "h", "i", "j"));
        out.add("setOf.arities=" + Set.of().size() + Set.of("a").size() + Set.of("a", "b").size()
            + Set.of("a", "b", "c").size() + Set.of("a", "b", "c", "d").size()
            + Set.of("a", "b", "c", "d", "e").size()
            + Set.of("a", "b", "c", "d", "e", "f").size()
            + Set.of("a", "b", "c", "d", "e", "f", "g").size()
            + Set.of("a", "b", "c", "d", "e", "f", "g", "h").size()
            + Set.of("a", "b", "c", "d", "e", "f", "g", "h", "i").size()
            + Set.of("a", "b", "c", "d", "e", "f", "g", "h", "i", "j").size());

        Set<String> set = Set.of("b", "a");
        out.add("set.sorted=" + sorted(set));
        out.add("set.size=" + set.size());
        out.add("set.contains=" + set.contains("a"));
        out.add("set.equals=" + set.equals(new java.util.HashSet<>(java.util.Arrays.asList("a", "b"))));
        out.add("set.add=" + attempt(() -> set.add("c")));
        out.add("set.ofNull=" + attempt(() -> Set.of((String) null)));
        out.add("set.ofDuplicate=" + attempt(() -> Set.of("a", "a")));

        Map<String, Integer> map = Map.of("b", 2, "a", 1);
        out.add("map.size=" + map.size());
        out.add("map.get=" + map.get("a") + "," + map.get("b"));
        out.add("map.sortedKeys=" + sorted(map.keySet()));
        out.add("map.put=" + attempt(() -> map.put("c", 3)));
        out.add("map.ofNullValue=" + attempt(() -> Map.of("k", (Integer) null)));
        out.add("map.ofDuplicateKey=" + attempt(() -> Map.of("k", 1, "k", 2)));
        out.add("map.empty=" + Map.of());

        Map.Entry<String, Integer> entry = Map.entry("k", 1);
        out.add("entry=" + entry);
        out.add("entry.setValue=" + attempt(() -> entry.setValue(2)));
        out.add("entry.ofNull=" + attempt(() -> Map.entry("k", (Integer) null)));

        out.add("optional.isEmpty=" + Optional.empty().isEmpty() + "," + Optional.of("x").isEmpty());
        out.add("optional.orElseThrow=" + Optional.of("x").orElseThrow());
        out.add("optional.orElseThrowEmpty=" + attempt(() -> Optional.empty().orElseThrow()));

        // ofNullable is the JDK's own null check: empty() for null, and
        // otherwise the same single-element stream of(T) builds.
        out.add("ofNullable=" + Stream.ofNullable("x").collect(Collectors.toList()));
        out.add("ofNullableNull=" + Stream.ofNullable(null).collect(Collectors.toList()));
        out.add("ofNullable.count=" + Stream.ofNullable("x").count()
            + "," + Stream.ofNullable(null).count());
        // A stream is consumed once, so a branch returning a shared constant
        // rather than a fresh stream would show up here.
        Stream<String> reused = Stream.ofNullable("x");
        out.add("ofNullable.firstUse=" + reused.count());
        out.add("ofNullable.secondUse=" + attempt(() -> reused.count()));
        out.add("ofNullable.nullElement=" + Stream.of((String) null).count());

        // Method references name their target in a bootstrap argument rather
        // than in an instruction. Both rewrite kinds have to be reachable that
        // way: an inline rebuild generates a helper for the handle to point at,
        // and a runtime one points the handle at the injected utility.
        java.util.function.Predicate<Optional<String>> isEmpty = Optional::isEmpty;
        out.add("methodRef.optionalIsEmpty=" + isEmpty.test(Optional.empty())
            + "," + isEmpty.test(Optional.of("x")));

        java.util.function.Supplier<List<String>> emptyList = List::of;
        out.add("methodRef.listOf=" + emptyList.get());
        out.add("methodRef.listOfIsUnmodifiable=" + attempt(() -> emptyList.get().add("z")));

        java.util.function.Function<String, String> strip = String::strip;
        out.add("methodRef.strip=[" + strip.apply("  a  ") + "]");

        // These return a Collector, so the replacement has to be one too. The
        // Java 8 shapes underneath, toSet and toMap, accept nulls where these
        // refuse them, so the checks are the whole point of the rewrite.
        out.add("toUnmodifiableSet=" + sorted(
            Stream.of("b", "a", "b").collect(Collectors.toUnmodifiableSet())));
        out.add("toUnmodifiableSet.isUnmodifiable=" + attempt(() ->
            Stream.of("a").collect(Collectors.toUnmodifiableSet()).add("z")));
        out.add("toUnmodifiableSet.nullElement=" + attempt(() ->
            Stream.of("a", (String) null).collect(Collectors.toUnmodifiableSet())));

        Map<String, Integer> lengths =
            Stream.of("a", "bb").collect(Collectors.toUnmodifiableMap(s -> s, String::length));
        out.add("toUnmodifiableMap=" + lengths.get("a") + "," + lengths.get("bb"));
        out.add("toUnmodifiableMap.isUnmodifiable=" + attempt(() -> lengths.put("c", 3)));
        // The message, not just the type: Java 8 toMap reports the value where
        // Java 9 onwards reports the key (JDK-8040892).
        out.add("toUnmodifiableMap.duplicateKey=" + detail(() ->
            Stream.of("a", "a").collect(Collectors.toUnmodifiableMap(s -> s, x -> "V" + x))));
        out.add("toUnmodifiableMap.nullKey=" + attempt(() ->
            Stream.of("a").collect(Collectors.toUnmodifiableMap(s -> (String) null, String::length))));
        out.add("toUnmodifiableMap.nullValue=" + attempt(() ->
            Stream.of("a").collect(Collectors.toUnmodifiableMap(s -> s, s -> (Integer) null))));

        Map<String, Integer> merged = Stream.of("a", "a", "b")
            .collect(Collectors.toUnmodifiableMap(s -> s, String::length, (x, y) -> x + y));
        out.add("toUnmodifiableMap.merged=" + merged.get("a") + "," + merged.get("b"));
        out.add("toUnmodifiableMap.mergedIsUnmodifiable=" + attempt(() -> merged.put("c", 3)));
        // Refused when the collector is built, before any element is seen, and
        // the JDK names which argument was null.
        out.add("toUnmodifiableMap.nullKeyMapper=" + detail(() ->
            Collectors.toUnmodifiableMap(null, s -> s)));
        out.add("toUnmodifiableMap.nullValueMapper=" + detail(() ->
            Collectors.toUnmodifiableMap(s -> s, null)));
        // Refused when the collector is built, before any element is seen.
        out.add("toUnmodifiableMap.nullMerge=" + detail(() ->
            Collectors.<String, String, String>toUnmodifiableMap(
                s -> s, s -> s, (BinaryOperator<String>) null)));

        out.add("toUnmodifiableList=" +
            Stream.of("b", "a", "b").collect(Collectors.toUnmodifiableList()));
        out.add("toUnmodifiableList.isUnmodifiable=" + attempt(() ->
            Stream.of("a").collect(Collectors.toUnmodifiableList()).add("z")));
        out.add("toUnmodifiableList.nullElement=" + attempt(() ->
            Stream.of("a", (String) null).collect(Collectors.toUnmodifiableList())));

        // not() is negate() plus a null check, and negate has been a default
        // method since 8, so the lambda it stands for already exists.
        Predicate<String> empty = String::isEmpty;
        out.add("predicateNot=" + Predicate.not(empty).test("")
            + "," + Predicate.not(empty).test("x"));
        out.add("predicateNot.filters=" + Stream.of("a", "", "b")
            .filter(Predicate.not(empty)).collect(Collectors.toList()));
        out.add("predicateNot.twice=" + Predicate.not(Predicate.not(empty)).test(""));
        out.add("predicateNot.null=" + attempt(() -> Predicate.not((Predicate<String>) null)));

        // All three return the index they were handed, so they read inside an
        // expression, and all three throw the same type. Only the message tells
        // them apart, so the message is what is checked.
        out.add("checkIndex=" + Objects.checkIndex(2, 5));
        out.add("checkFromToIndex=" + Objects.checkFromToIndex(1, 3, 5));
        out.add("checkFromIndexSize=" + Objects.checkFromIndexSize(1, 2, 5));

        out.add("checkIndex.over=" + detail(() -> Objects.checkIndex(5, 3)));
        out.add("checkIndex.negative=" + detail(() -> Objects.checkIndex(-1, 3)));
        // A half-open range, so an empty one at zero length is fine.
        out.add("checkIndex.emptyLength=" + detail(() -> Objects.checkIndex(0, 0)));
        out.add("checkFromToIndex.empty=" + detail(() -> Objects.checkFromToIndex(0, 0, 0)));
        out.add("checkFromToIndex.reversed=" + detail(() -> Objects.checkFromToIndex(2, 1, 5)));
        out.add("checkFromToIndex.past=" + detail(() -> Objects.checkFromToIndex(0, 6, 5)));
        out.add("checkFromToIndex.negative=" + detail(() -> Objects.checkFromToIndex(-1, 2, 5)));
        out.add("checkFromIndexSize.past=" + detail(() -> Objects.checkFromIndexSize(3, 4, 5)));
        out.add("checkFromIndexSize.negativeFrom=" + detail(() -> Objects.checkFromIndexSize(-1, 1, 5)));
        out.add("checkFromIndexSize.negativeSize=" + detail(() -> Objects.checkFromIndexSize(1, -1, 5)));
        // fromIndex + size overflows here, so a naive sum would pass this.
        out.add("checkFromIndexSize.overflow=" + detail(() ->
            Objects.checkFromIndexSize(Integer.MAX_VALUE - 1, 5, Integer.MAX_VALUE)));

        out.add("repeat=" + "ab".repeat(3));
        out.add("repeatZero=[" + "ab".repeat(0) + "]");
        out.add("repeatNegative=" + attempt(() -> "ab".repeat(-1)));
        // length * count overflows to a negative capacity, so the guard has to
        // come before the allocation rather than out of it.
        out.add("repeatOverflow=" + detail(() -> "ab".repeat(Integer.MAX_VALUE)));
        out.add("repeatOne=" + "ab".repeat(1));
        out.add("repeatEmptyValue=[" + "".repeat(5) + "]");

        out.add("requireNonNullElse=" + Objects.requireNonNullElse("a", "b")
            + "," + Objects.requireNonNullElse(null, "b"));
        out.add("requireNonNullElseBothNull=" + attempt(() -> Objects.requireNonNullElse(null, null)));

        // Past ten elements javac switches to the varargs overload.
        List<String> bigList = List.of("a","b","c","d","e","f","g","h","i","j","k");
        out.add("bigList=" + bigList);
        out.add("bigList.add=" + attempt(() -> bigList.add("z")));
        out.add("bigList.ofNull=" + attempt(
            () -> List.of("a","b","c","d","e","f","g","h","i","j",(String) null)));

        Set<String> bigSet = Set.of("a","b","c","d","e","f","g","h","i","j","k");
        out.add("bigSet.size=" + bigSet.size());
        out.add("bigSet.sorted=" + sorted(bigSet));
        out.add("bigSet.add=" + attempt(() -> bigSet.add("z")));
        out.add("bigSet.ofDuplicate=" + attempt(
            () -> Set.of("a","b","c","d","e","f","g","h","i","j","a")));
        out.add("bigSet.ofNull=" + attempt(
            () -> Set.of("a","b","c","d","e","f","g","h","i","j",(String) null)));

        Map<String, String> entries = Map.ofEntries(Map.entry("a", "1"), Map.entry("b", "2"));
        out.add("ofEntries.size=" + entries.size());
        out.add("ofEntries.get=" + entries.get("a") + "," + entries.get("b"));
        out.add("ofEntries.put=" + attempt(() -> entries.put("c", "3")));
        out.add("ofEntries.duplicateKey=" + attempt(
            () -> Map.ofEntries(Map.entry("k", "1"), Map.entry("k", "2"))));
        // Map.entry refuses a null of its own, so reaching ofEntries' checks
        // needs an entry type that does not.
        out.add("ofEntries.nullValue=" + attempt(
            () -> Map.ofEntries(new AbstractMap.SimpleEntry<String, String>("k", null))));
        out.add("ofEntries.nullKey=" + attempt(
            () -> Map.ofEntries(new AbstractMap.SimpleEntry<String, String>(null, "v"))));
        out.add("ofEntries.nullEntry=" + attempt(
            () -> Map.ofEntries((Map.Entry<String, String>) null)));
        out.add("ofEntries.empty=" + Map.ofEntries());

        // copyOf snapshots, and unlike Set.of it drops a repeat rather than
        // refusing it.
        List<String> mutable = new ArrayList<>(Arrays.asList("x", "y"));
        List<String> listCopy = List.copyOf(mutable);
        Set<String> setCopy = Set.copyOf(Arrays.asList("a", "a", "b"));
        mutable.add("z");
        out.add("listCopy=" + listCopy);
        out.add("listCopy.isSnapshot=" + (listCopy.size() == 2));
        out.add("listCopy.add=" + attempt(() -> listCopy.add("z")));
        out.add("listCopy.ofNullElement=" + attempt(() -> List.copyOf(Arrays.asList("a", null))));

        // The copy has to be taken before the null check, not after it.
        Leaky leaky = new Leaky();
        List<String> defensive = List.copyOf(leaky);
        Set<String> defensiveSet = Set.copyOf(leaky);
        leaky.shared[0] = null;
        out.add("listCopy.defensive=" + defensive);
        out.add("setCopy.defensive=" + sorted(defensiveSet));
        out.add("setCopy.size=" + setCopy.size());
        out.add("setCopy.sorted=" + sorted(setCopy));
        out.add("setCopy.repeatsDropped=ok");

        Map<String, String> sourceMap = new LinkedHashMap<>();
        sourceMap.put("a", "1");
        Map<String, String> mapCopy = Map.copyOf(sourceMap);
        sourceMap.put("b", "2");
        // Map.entrySet is Java 8 and must survive untouched. Rewriting Map.entry
        // on a name prefix would take it too, so the call is made here rather
        // than left to turn up incidentally inside some rewrite's own output.
        out.add("entrySet.size=" + sourceMap.entrySet().size());
        out.add("entrySet.first=" + sourceMap.entrySet().iterator().next());

        out.add("mapCopy.size=" + mapCopy.size());
        out.add("mapCopy.put=" + attempt(() -> mapCopy.put("c", "3")));
        Map<String, String> withNullValue = new LinkedHashMap<>();
        withNullValue.put("k", null);
        out.add("mapCopy.nullValue=" + attempt(() -> Map.copyOf(withNullValue)));

        return String.join("\n", out);
    }

    public static void main(String[] args) {
        System.out.println(runAll());
    }
}
