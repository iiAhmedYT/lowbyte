import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.List;

/**
 * The `Arrays` comparison methods Java 9 added, all fifty-nine overloads.
 *
 * Every overload is called at least once, and that is the point of the sample
 * rather than a side effect of being thorough. A rewrite is matched on its exact
 * descriptor, so one wrong character means the call is not rewritten at all and
 * dies on Java 8 with a `NoSuchMethodError`. Only calling it says otherwise.
 *
 * Past that the cases are the ones where a plausible implementation and the real
 * one part company: a prefix, an empty range, NaN, -0.0, an unsigned byte, a
 * null array, a reversed range, and a comparator counting how often it is asked.
 */
public class ArraysSample {

    private interface Call {
        Object run();
    }

    /**
     * The value, or what was thrown trying to get it.
     *
     * A null pointer keeps only its type: the message naming the variable is a
     * Java 14 feature, so it cannot survive a downgrade and is not supposed to.
     * The other two carry their message, which is wording we owe the caller.
     */
    private static String attempt(Call call) {
        try {
            return String.valueOf(call.run());
        } catch (NullPointerException e) {
            return "NullPointerException";
        } catch (Throwable t) {
            return t.getClass().getSimpleName() + "(" + t.getMessage() + ")";
        }
    }

    private static void ints(List<String> out) {
        final int[] empty = {};
        final int[] a = { 1, 2, 3 };
        final int[] b = { 1, 2, 3 };
        final int[] c = { 1, 2, 4 };
        final int[] prefix = { 1, 2 };
        final int[] minus = { -1 };
        final int[] one = { 1 };
        final int[] seven = new int[7];
        final int[] two = new int[2];

        out.add("int mismatch(a,b)=" + Arrays.mismatch(a, b));
        out.add("int mismatch(a,c)=" + Arrays.mismatch(a, c));
        out.add("int mismatch(a,prefix)=" + Arrays.mismatch(a, prefix));

        out.add("int equals(a,1,1,c,2,2)=" + Arrays.equals(a, 1, 1, c, 2, 2));
        out.add("int compare(a,b)=" + Arrays.compare(a, b));
        out.add("int compare(a,c)=" + Arrays.compare(a, c));
        out.add("int compare(seven,two)=" + Arrays.compare(seven, two));
        out.add("int compare(empty,two)=" + Arrays.compare(empty, two));
        out.add("int compare(null,null)=" + Arrays.compare((int[]) null, (int[]) null));
        out.add("int compare(null,empty)=" + Arrays.compare((int[]) null, empty));
        out.add("int compare(a,1,3,c,1,3)=" + Arrays.compare(a, 1, 3, c, 1, 3));
        out.add("int compareUnsigned(minus,one)=" + Arrays.compareUnsigned(minus, one));
        out.add("int compareUnsigned(one,minus)=" + Arrays.compareUnsigned(one, minus));
        out.add("int compareUnsigned(null,null)=" + Arrays.compareUnsigned((int[]) null, (int[]) null));
        out.add("int compareUnsigned(minus,0,1,one,0,1)=" + Arrays.compareUnsigned(minus, 0, 1, one, 0, 1));

        // The shared bounds check, and the order it applies its two exceptions.
        out.add("int mismatch(a,2,1,..)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch(a, 2, 1, c, 0, 3); }
        }));
        out.add("int mismatch(a,-1,3,..)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch(a, -1, 3, c, 0, 3); }
        }));
        out.add("int mismatch(a,0,9,..)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch(a, 0, 9, c, 0, 3); }
        }));
        out.add("int mismatch(a,-1,-1,..)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch(a, -1, -1, c, 0, 3); }
        }));
        // Negative and reversed at once, which is the only case that says which
        // of the two checks runs first.
        out.add("int mismatch(a,-1,-2,..)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch(a, -1, -2, c, 0, 3); }
        }));
        out.add("int mismatch(a,0,3,c,0,9)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch(a, 0, 3, c, 0, 9); }
        }));
        out.add("int mismatch(null,null)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch((int[]) null, (int[]) null); }
        }));
        // Checked even where the differing lengths already settle the answer.
        out.add("int equals(a,0,3,c,0,9)=" + attempt(new Call() {
            public Object run() { return Arrays.equals(a, 0, 3, c, 0, 9); }
        }));
        out.add("int compare(a,0,9,c,0,3)=" + attempt(new Call() {
            public Object run() { return Arrays.compare(a, 0, 9, c, 0, 3); }
        }));
        out.add("int compareUnsigned(a,0,9,c,0,3)=" + attempt(new Call() {
            public Object run() { return Arrays.compareUnsigned(a, 0, 9, c, 0, 3); }
        }));
    }

    /** A comparator that counts, because how often it is asked is observable. */
    private static final class Counting implements Comparator<String> {

        private int calls;

        public int compare(String left, String right) {
            calls++;
            if (left == null || right == null) {
                return left == null ? (right == null ? 0 : -1) : 1;
            }
            return left.compareTo(right);
        }
    }

    private static void comparators(List<String> out) {
        final String shared = "x";
        final String[] a = { shared, shared, shared };
        final String[] copy = { shared, shared, shared };
        final String[] other = { shared, shared, "y" };
        final String[] prefix = { shared, shared };

        Counting counter = new Counting();
        out.add("cmp2 mismatch(a,copy)=" + Arrays.mismatch(a, copy, counter));
        out.add("cmp2 mismatch calls=" + counter.calls);

        counter = new Counting();
        out.add("cmp2 mismatch(a,a)=" + Arrays.mismatch(a, a, counter));
        out.add("cmp2 mismatch(a,a) calls=" + counter.calls);

        counter = new Counting();
        out.add("cmp2 mismatch(a,other)=" + Arrays.mismatch(a, other, counter));
        out.add("cmp2 mismatch(a,prefix)=" + Arrays.mismatch(a, prefix, counter));

        counter = new Counting();
        out.add("cmp2 mismatch(a,0,3,other,0,3)=" + Arrays.mismatch(a, 0, 3, other, 0, 3, counter));
        out.add("cmp2 mismatch range calls=" + counter.calls);

        // equals asks about two identical references and mismatch does not, so
        // these two counts are supposed to disagree.
        counter = new Counting();
        out.add("cmp2 equals(a,copy)=" + Arrays.equals(a, copy, counter));
        out.add("cmp2 equals calls=" + counter.calls);

        counter = new Counting();
        out.add("cmp2 equals(a,a)=" + Arrays.equals(a, a, counter));
        out.add("cmp2 equals(a,a) calls=" + counter.calls);

        counter = new Counting();
        out.add("cmp2 equals(a,prefix)=" + Arrays.equals(a, prefix, counter));
        out.add("cmp2 equals(a,prefix) calls=" + counter.calls);
        out.add("cmp2 equals(a,null)=" + Arrays.equals(a, (String[]) null, counter));
        out.add("cmp2 equals(null,null)=" + Arrays.equals((String[]) null, (String[]) null, counter));

        counter = new Counting();
        out.add("cmp2 equals(a,0,3,copy,0,3)=" + Arrays.equals(a, 0, 3, copy, 0, 3, counter));
        out.add("cmp2 equals range calls=" + counter.calls);
        out.add("cmp2 equals(a,0,3,prefix,0,2)=" + Arrays.equals(a, 0, 3, prefix, 0, 2, counter));

        counter = new Counting();
        out.add("cmp2 compare(a,copy)=" + Arrays.compare(a, copy, counter));
        out.add("cmp2 compare calls=" + counter.calls);
        out.add("cmp2 compare(a,other)=" + Arrays.compare(a, other, counter));
        out.add("cmp2 compare(a,prefix)=" + Arrays.compare(a, prefix, counter));
        out.add("cmp2 compare(null,null)=" + Arrays.compare((String[]) null, (String[]) null, counter));
        out.add("cmp2 compare(null,a)=" + Arrays.compare((String[]) null, a, counter));
        out.add("cmp2 compare(a,null)=" + Arrays.compare(a, (String[]) null, counter));
        out.add("cmp2 compare(a,0,3,other,0,3)=" + Arrays.compare(a, 0, 3, other, 0, 3, counter));

        // The comparator is checked before either array is looked at, so a bad
        // range does not get the chance to throw first.
        out.add("cmp2 compare(a,0,9,..,null)=" + attempt(new Call() {
            public Object run() { return Arrays.compare(a, 0, 9, other, 0, 3, null); }
        }));
        out.add("cmp2 mismatch(a,0,9,..,null)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch(a, 0, 9, other, 0, 3, null); }
        }));
        out.add("cmp2 equals(a,0,9,..,null)=" + attempt(new Call() {
            public Object run() { return Arrays.equals(a, 0, 9, other, 0, 3, null); }
        }));
        out.add("cmp2 mismatch(a,null,cmp)=" + attempt(new Call() {
            public Object run() { return Arrays.mismatch(a, (String[]) null, new Counting()); }
        }));
    }

    public static String runAll() {
        List<String> out = new ArrayList<String>();
        bytes(out);
        chars(out);
        doubles(out);
        floats(out);
        ints(out);
        longs(out);
        shorts(out);
        booleans(out);
        references(out);
        comparables(out);
        comparators(out);
        return String.join("\n", out);
    }

    public static void main(String[] args) {
        System.out.println(runAll());
    }
